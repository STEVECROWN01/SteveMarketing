    65→    65→
    66→    66→export function LeadForm({ kind }: { kind: FormKind }) {
    67→    67→  const isQuote = kind === "quote";
    68→    68→  const isContact = kind === "contact";
    69→    69→  const isLight = isQuote;
    70→    70→  const [form, setForm] = useState<FormState>(EMPTY);
    71→    71→  const [errors, setErrors] = useState<Record<string, string>>({});
    72→    72→  const [submitting, setSubmitting] = useState(false);
    73→    73→  const [done, setDone] = useState(false);
    74→    74→  const formRef = useRef<HTMLFormElement>(null);
    75→    75→  const mountedAt = useRef<number>(Date.now());
    76→    76→  const { toast } = useToast();
    77→    77→  const router = useRouter();
    78→    78→
    79→    79→  const set = (key: keyof FormState) => (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement | HTMLSelectElement>) => {
    80→    80→    setForm((f) => ({ ...f, [key]: e.target.value }));
    81→    81→    setErrors((prev) => {
    82→    82→      if (!prev[key]) return prev;
    83→    83→      const next = { ...prev };
    84→    84→      delete next[key];
    85→    85→      return next;
    86→    86→    });
    87→    87→  };
    88→    88→
    89→    89→  function validate(): Record<string, string> {
    90→    90→    const errs: Record<string, string> = {};
    91→    91→    if (form.name.trim().length < 2) errs.name = "Indiquez votre nom complet.";
    92→    92→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(form.email.trim()))
    93→    93→      errs.email = "Indiquez une adresse email valide.";
    94→    94→    if (!isQuote && !isContact && form.website.trim().length < 4)
    95→    95→      errs.website = "L'URL de votre site actuel est nécessaire pour l'audit.";
    96→    96→    if (form.phone.trim() && !/^[+\d][\d\s().-]{5,}$/.test(form.phone.trim()))
    97→    97→      errs.phone = "Numéro de téléphone invalide.";
    98→    98→    if (isContact && form.phone.trim().length < 6)
    99→    99→      errs.phone = "Indiquez votre numéro de téléphone.";
   100→   100→    if (isContact && form.company.trim().length < 2)
   101→   101→      errs.company = "Indiquez le nom de votre entreprise ou activité.";
   102→   102→    if (isContact && form.discovery === "")
   103→   103→      errs.discovery = "Indiquez comment vous m'avez découvert.";
   104→   104→    if (form.message.trim().length < 10)
   105→   105→      errs.message = "Décrivez votre besoin en quelques mots (10 caractères minimum).";
   106→   106→    if (isQuote && form.serviceType === "") errs.serviceType = "Choisissez le service souhaité.";
   107→   107→    return errs;
   108→   108→  }
   109→   109→
   110→   110→  async function handleSubmit(e: React.FormEvent) {
   111→   111→    e.preventDefault();
   112→   112→    if (submitting) return;
   113→   113→
   114→   114→    // Anti-spam : leurre invisible
   115→   115→    if (form.honeypot) return;
   116→   116→
   117→   117→    // Anti-spam : soumission trop rapide (< 3 secondes)
   118→   118→    if (Date.now() - mountedAt.current < 3000) {

   120→   120→      return;
   121→   121→    }
   122→   122→
   123→   123→    const clientErrors = validate();
   124→   124→    if (Object.keys(clientErrors).length > 0) {
   125→   125→      setErrors(clientErrors);
   126→   126→      const first = document.getElementById(`${kind}-field-${Object.keys(clientErrors)[0]}`);
   127→   127→      first?.focus();

   129→   129→    }
   130→   130→
   131→   131→    setSubmitting(true);
   132→   132→    setErrors({});
   133→   133→
   134→   134→    const payload: Partial<LeadInput> = {
   135→   135→      type: kind,
   136→   136→      name: form.name.trim(),
   137→   137→      email: form.email.trim(),
   138→   138→      phone: form.phone.trim() || undefined,
   139→   139→      company: form.company.trim() || undefined,
   140→   140→      website: form.website.trim() || undefined,
   141→   141→      activity: form.activity.trim() || undefined,
   142→   142→      discovery: isContact ? form.discovery : undefined,
   143→   143→      serviceType: isQuote ? form.serviceType : undefined,
   144→   144→      budget: isQuote ? form.budget || undefined : undefined,
   145→   145→      timeline: isQuote ? form.timeline || undefined : undefined,
   146→   146→      message: form.message.trim(),
   147→   147→      honeypot: "",
   148→   148→    };
   149→   149→
   150→   150→    try {
   151→   151→      const res = await fetch("/api/leads", {
   152→   152→        method: "POST",
   153→   153→        headers: { "Content-Type": "application/json" },
   154→   154→        body: JSON.stringify(payload),
   155→   155→      });
   156→   156→      const data = (await res.json().catch(() => null)) as
   157→   157→        | { ok: boolean; id?: string; error?: string; fieldErrors?: Record<string, string> }
   158→   158→        | null;
   159→   159→
   160→   160→      if (!res.ok || !data?.ok) {
   161→   161→        if (data?.fieldErrors) setErrors(data.fieldErrors);
   162→   162→        throw new Error(data?.error ?? "Erreur inconnue");
   163→   163→      }
   164→   164→
   165→   165→      setForm(EMPTY);
   166→   166→
   167→   167→      if (isContact) {
   168→   168→        // Message de contact : redirection vers la page dédiée à la
   169→   169→        // réservation d'appel (demande de l'éditeur, 13/11/2026) —
   170→   170→        // les informations ont déjà été transmises par email.
   171→   171→        toast({
   172→   172→          title: "Message envoyé",
   173→   173→          description:
   174→   174→            "Merci ! Votre message est bien parti. Réservez maintenant votre créneau d'appel stratégique.",
   175→   175→        });
   176→   176→        router.navigate("appel");
   177→   177→        return;
   178→   178→      }
   179→   179→
   180→   180→      setDone(true);
   181→   181→      toast({
   182→   182→        title: "Demande enregistrée",
   183→   183→        description:
   184→   184→          "Votre demande a bien été envoyée. Un accusé de réception vous est adressé par email.",
   185→   185→      });
   186→   186→      // Confirmation visible après enregistrement effectif
   187→   187→      document
   188→   188→        .getElementById(`${kind}-confirmation`)
   189→   189→        ?.scrollIntoView({ behavior: "smooth", block: "center" });
   190→   190→    } catch (err) {
   191→   191→      const message =
   192→   192→        err instanceof Error && err.message !== "Erreur inconnue"
   193→   193→          ? err.message
   194→   194→          : `Une erreur est survenue pendant l'envoi. Merci de réessayer ou de m'écrire directement à ${SITE.email}.`;
   195→   195→      setErrors({ form: message });
   196→   196→      toast({
   197→   197→        title: "Envoi impossible",
   198→   198→        description: message,
   199→   199→        variant: "destructive",
   200→   200→      });
   201→   201→    } finally {
   202→   202→      setSubmitting(false);
   203→   203→    }
   204→   204→  }
   205→   205→
   206→   206→  /* ---------- Styles par thème ---------- */

   208→   208→    ? "min-h-[48px] w-full rounded-lg border border-[#e4e4e4] bg-white px-4 py-3 text-[15px] text-[#1c1c1c] placeholder:text-[#525252]/60 transition-colors focus:border-[#1c1c1c] focus:outline-none"
   209→   209→    : isContact
   210→   210→      ? "min-h-[50px] w-full rounded-md bg-white px-4 py-3 text-[15px] text-[#1c1c1c] placeholder:text-[#525252]/70 transition-shadow duration-200 focus:outline-none focus:shadow-[0_0_0_2px_rgba(228,228,228,0.55)]"
   211→   211→      : "min-h-[48px] w-full rounded-lg border border-[#e4e4e4]/18 bg-[#1c1c1c] px-4 py-3 text-[15px] text-white placeholder:text-[#c8c8c8]/50 transition-colors focus:border-[#e4e4e4] focus:outline-none";
   212→   212→  const labelBase = isLight
   213→   213→    ? "mb-1.5 block text-sm font-medium text-[#1c1c1c]"

   215→   215→  const errorBase = "mt-1.5 text-xs text-[#ea384c]";
   216→   216→
   217→   217→  /* ---------- Confirmation (devis / audit) ---------- */
   218→   218→  if (done) {
   219→   219→    return (
   220→   220→      <div
   221→   221→        id={`${kind}-confirmation`}
   222→   222→        className={
   223→   223→          isLight
   224→   224→            ? "rounded-2xl border border-[#e4e4e4] bg-white p-8 text-center sm:p-10"
   225→   225→            : "rounded-2xl border border-[#e4e4e4]/40 bg-[#232323] p-8 text-center sm:p-10"
   226→   226→        }
   227→   227→      >
   228→   228→        <span
   229→   229→          className={
   230→   230→            isLight
   231→   231→              ? "mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-[#1c1c1c] text-white"
   232→   232→              : "mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-[#e4e4e4] text-[#1c1c1c]"
   233→   233→          }
   234→   234→        >
   235→   235→          <CheckCircle2 className="h-7 w-7" aria-hidden="true" />
   236→   236→        </span>

   239→   239→        >
   240→   240→          Votre demande a bien été enregistrée.
   241→   241→        </h3>
   242→   242→        <p className={`mx-auto mt-4 max-w-md text-[15px] leading-relaxed ${isLight ? "text-[#525252]" : "text-[#c8c8c8]"}`}>
   243→   243→          {isQuote
   244→   244→            ? "Merci pour votre confiance. Je reviens vers vous très rapidement pour préparer votre devis personnalisé."
   245→   245→            : "Merci pour votre confiance. Je reviens vers vous très rapidement avec les premières observations sur votre site."}
   246→   246→        </p>
   247→   247→        <p className={`mt-4 text-sm ${isLight ? "text-[#525252]" : "text-[#c8c8c8]"}`}>
   248→   248→          Un accusé de réception vous est adressé par email.
   249→   249→        </p>
   250→   250→        <div className="mt-8 flex flex-col items-center justify-center gap-3 sm:flex-row">
   251→   251→          <button
   252→   252→            type="button"
   253→   253→            onClick={() => {
   254→   254→              setDone(false);
   255→   255→              mountedAt.current = Date.now();
   256→   256→            }}
   257→   257→            className={
   258→   258→              isLight
   259→   259→                ? "btn-fill min-h-[44px] rounded-full border border-[#1c1c1c]/25 bg-transparent px-6 py-3 text-sm font-medium text-[#1c1c1c] [&::before]:bg-[#e4e4e4]"
   260→   260→                : "btn-fill min-h-[44px] rounded-full border border-[#e4e4e4]/30 px-6 py-3 text-sm font-medium text-white [&::before]:bg-[#3c3c3c]"
   261→   261→            }
   262→   262→          >
   263→   263→            <span className="inline-flex items-center">Envoyer Une Autre Demande</span>
   264→   264→          </button>
   265→   265→        </div>
   266→   266→        <p className={`mt-6 text-xs ${isLight ? "text-[#525252]/80" : "text-[#c8c8c8]/70"}`}>
   267→   267→          Vous préférez avancer plus vite&nbsp;?{" "}
   268→   268→          <a
   269→   269→            href={SITE.calendly}
   270→   270→            target="_blank"
   271→   271→            rel="noopener noreferrer"
   272→   272→            className={`font-medium underline underline-offset-4 ${isLight ? "text-[#1c1c1c]" : "text-[#e4e4e4]"}`}
   273→   273→          >
   274→   274→            Réservez un appel de 30 minutes
   275→   275→          </a>{" "}
   276→   276→          — {SITE.calendlyMention.toLowerCase()}
   277→   277→        </p>
   278→   278→      </div>
   279→   279→    );
   280→   280→  }
   281→   281→
   282→   282→  /* ---------- Formulaire ---------- */
   283→   283→  return (
   284→   284→    <form
   285→   285→      ref={formRef}
   286→   286→      onSubmit={handleSubmit}
   287→   287→      noValidate
   288→   288→      aria-label={
   289→   289→        isContact
   290→   290→          ? "Formulaire de contact — laissez un message"
   291→   291→          : isQuote
   292→   292→            ? "Formulaire de demande de devis personnalisé"
   293→   293→            : "Formulaire de demande d'audit gratuit"
   294→   294→      }
   295→   295→      className={
   296→   296→        isQuote
   297→   297→          ? // Page claire (#/demander-un-devis) : champs posés directement
   298→   298→            // sur le fond blanc, façon sheragency.
   299→   299→            "relative"
   300→   300→          : isContact
   301→   301→            ? // Page sombre (#/contact) : champs posés directement sur le
   302→   302→              // fond sombre, entrées blanches, façon sheragency.
   303→   303→              "relative"
   304→   304→            : "rounded-2xl border border-[#e4e4e4]/12 bg-[#232323] p-6 sm:p-8 lg:p-9"
   305→   305→      }
   306→   306→    >
   307→   307→      {/* Piège anti-robots (invisible) */}
   308→   308→      <input
   309→   309→        type="text"
   310→   310→        name="societe_verification"
   311→   311→        value={form.honeypot}
   312→   312→        onChange={set("honeypot")}
   313→   313→        tabIndex={-1}
   314→   314→        autoComplete="off"
   315→   315→        aria-hidden="true"
   316→   316→        className="pointer-events-none absolute left-[-9999px] h-0 w-0 opacity-0"
   317→   317→      />
   318→   318→
   319→   319→      {/* ------- Variante « contact » : champs empilés (sheragency) ------- */}

   321→   321→        <div className="flex flex-col gap-5">
   322→   322→          <div>

   335→   335→              className={inputBase}

   337→   337→            />

   343→   343→          </div>

   345→   345→          <div>
   346→   346→            <label htmlFor={`${kind}-field-phone`} className={labelBase}>
   347→   347→              Téléphone <span className={isLight ? "text-[#1c1c1c]" : "text-[#e4e4e4]"}>*</span>
   348→   348→            </label>
   349→   349→            <input

   362→   362→            />
   363→   363→            {errors.phone && (

   368→   368→          </div>

   370→   370→          <div>

   395→   395→          <div>

   410→   410→            />
   411→   411→            {errors.company && (

   413→   413→                {errors.company}
   414→   414→              </p>

   416→   416→          </div>

   418→   418→          <div>
