     2→

     4→import { CheckCircle2, Loader2, AlertCircle } from "lucide-react";
     5→import { useToast } from "@/hooks/use-toast";
     6→import {
     7→  BUDGET_RANGES,

    15→
    16→/**
    17→ * Formulaire de demande — trois variantes (13/11/2026) :
    18→ * - « quote » (page #/demander-un-devis) : thème CLAIR, présentation
    19→ *   calquée sur la page « Request a quote » de sheragency.com — champs
    20→ *   d'identité en deux colonnes, message pleine largeur, bouton noir
    21→ *   pleine largeur. Confirmation inline.
    22→ * - « contact » (page #/contact) : thème SOMBRE, présentation calquée
    23→ *   sur « Contact Us » de sheragency.com — champs pleine largeur aux
    24→ *   entrées BLANCHES empilées, bouton blanc demi-largeur
    25→ *   « Envoyer Le Message ». Après envoi (email automatique à
    26→ *   stevensakpovi@gmail.com), redirection vers la page dédiée à la
    27→ *   réservation d'appel (#/reserver-un-appel).
    28→ * - « audit » (variante historique sombre, conservée pour référence).
    29→ * Validation frontend (contrôles) + backend (API), protection
    30→ * anti-spam (honeypot + délai), enregistrement fiable et accusé de
    31→ * réception. Aucun secret côté client.
    32→ */
    33→
    34→type FormKind = "quote" | "audit" | "contact";
    35→
    36→interface FormState {
    37→  name: string;
    38→  email: string;
    39→  phone: string;
    40→  company: string;
    41→  website: string;
    42→  activity: string;
    43→  discovery: string;
    44→  serviceType: string;
    45→  budget: string;
    46→  timeline: string;
    47→  message: string;
    48→  honeypot: string;
    49→}
    50→
    51→const EMPTY: FormState = {
    52→  name: "",
    53→  email: "",
    54→  phone: "",
    55→  company: "",
    56→  website: "",
    57→  activity: "",
    58→  discovery: "",
    59→  serviceType: "",
    60→  budget: "",
    61→  timeline: "",
    62→  message: "",
    63→  honeypot: "",
    64→};
    65→
    66→export function LeadForm({ kind }: { kind: FormKind }) {
    67→  const isQuote = kind === "quote";
    68→  const isContact = kind === "contact";
    69→  const isLight = isQuote;
    70→  const [form, setForm] = useState<FormState>(EMPTY);
    71→  const [errors, setErrors] = useState<Record<string, string>>({});
    72→  const [submitting, setSubmitting] = useState(false);
    73→  const [done, setDone] = useState(false);
    74→  const formRef = useRef<HTMLFormElement>(null);
    75→  const mountedAt = useRef<number>(Date.now());
    76→  const { toast } = useToast();
    77→  const router = useRouter();
    78→
    79→  const set = (key: keyof FormState) => (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement | HTMLSelectElement>) => {
    80→    setForm((f) => ({ ...f, [key]: e.target.value }));
    81→    setErrors((prev) => {
    82→      if (!prev[key]) return prev;
    83→      const next = { ...prev };
    84→      delete next[key];
    85→      return next;
    86→    });
    87→  };
    88→
    89→  function validate(): Record<string, string> {
    90→    const errs: Record<string, string> = {};
    91→    if (form.name.trim().length < 2) errs.name = "Indiquez votre nom complet.";
    92→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(form.email.trim()))
    93→      errs.email = "Indiquez une adresse email valide.";
    94→    if (!isQuote && !isContact && form.website.trim().length < 4)
    95→      errs.website = "L'URL de votre site actuel est nécessaire pour l'audit.";
    96→    if (form.phone.trim() && !/^[+\d][\d\s().-]{5,}$/.test(form.phone.trim()))
    97→      errs.phone = "Numéro de téléphone invalide.";
    98→    if (isContact && form.phone.trim().length < 6)
    99→      errs.phone = "Indiquez votre numéro de téléphone.";
   100→    if (isContact && form.company.trim().length < 2)
   101→      errs.company = "Indiquez le nom de votre entreprise ou activité.";
   102→    if (isContact && form.discovery === "")
   103→      errs.discovery = "Indiquez comment vous m'avez découvert.";
   104→    if (form.message.trim().length < 10)
   105→      errs.message = "Décrivez votre besoin en quelques mots (10 caractères minimum).";
   106→    if (isQuote && form.serviceType === "") errs.serviceType = "Choisissez le service souhaité.";
   107→    return errs;
   108→  }
   109→
   110→  async function handleSubmit(e: React.FormEvent) {
   111→    e.preventDefault();
   112→    if (submitting) return;
   113→
   114→    // Anti-spam : leurre invisible
   115→    if (form.honeypot) return;
   116→
   117→    // Anti-spam : soumission trop rapide (< 3 secondes)
   118→    if (Date.now() - mountedAt.current < 3000) {
   119→      setErrors({ form: "Prenez quelques instants pour compléter votre message, puis renvoyez." });
   120→      return;
   121→    }
   122→
   123→    const clientErrors = validate();
   124→    if (Object.keys(clientErrors).length > 0) {
   125→      setErrors(clientErrors);
   126→      const first = document.getElementById(`${kind}-field-${Object.keys(clientErrors)[0]}`);
   127→      first?.focus();
   128→      return;
   129→    }
   130→
   131→    setSubmitting(true);
   132→    setErrors({});
   133→
   134→    const payload: Partial<LeadInput> = {
   135→      type: kind,
   136→      name: form.name.trim(),
   137→      email: form.email.trim(),
   138→      phone: form.phone.trim() || undefined,
   139→      company: form.company.trim() || undefined,
   140→      website: form.website.trim() || undefined,
   141→      activity: form.activity.trim() || undefined,
   142→      discovery: isContact ? form.discovery : undefined,
   143→      serviceType: isQuote ? form.serviceType : undefined,
   144→      budget: isQuote ? form.budget || undefined : undefined,
   145→      timeline: isQuote ? form.timeline || undefined : undefined,
   146→      message: form.message.trim(),
   147→      honeypot: "",
   148→    };
   149→
   150→    try {
   151→      const res = await fetch("/api/leads", {
   152→        method: "POST",
   153→        headers: { "Content-Type": "application/json" },
   154→        body: JSON.stringify(payload),
   155→      });
   156→      const data = (await res.json().catch(() => null)) as
   157→        | { ok: boolean; id?: string; error?: string; fieldErrors?: Record<string, string> }
   158→        | null;
   159→
   160→      if (!res.ok || !data?.ok) {
   161→        if (data?.fieldErrors) setErrors(data.fieldErrors);
   162→        throw new Error(data?.error ?? "Erreur inconnue");
   163→      }
   164→
   165→      setForm(EMPTY);
   166→
   167→      if (isContact) {
   168→        // Message de contact : redirection vers la page dédiée à la
   169→        // réservation d'appel (demande de l'éditeur, 13/11/2026) —
   170→        // les informations ont déjà été transmises par email.
   171→        toast({
   172→          title: "Message envoyé",
   173→          description:
   174→            "Merci ! Votre message est bien parti. Réservez maintenant votre créneau d'appel stratégique.",
   175→        });
   176→        router.navigate("appel");
   177→        return;
   178→      }
   179→
   180→      setDone(true);
   181→      toast({
   182→        title: "Demande enregistrée",
   183→        description:
   184→          "Votre demande a bien été envoyée. Un accusé de réception vous est adressé par email.",
   185→      });
   186→      // Confirmation visible après enregistrement effectif
   187→      document
   188→        .getElementById(`${kind}-confirmation`)
   189→        ?.scrollIntoView({ behavior: "smooth", block: "center" });
   190→    } catch (err) {
   191→      const message =
   192→        err instanceof Error && err.message !== "Erreur inconnue"
   193→          ? err.message
   194→          : `Une erreur est survenue pendant l'envoi. Merci de réessayer ou de m'écrire directement à ${SITE.email}.`;
   195→      setErrors({ form: message });
   196→      toast({
   197→        title: "Envoi impossible",
   198→        description: message,
   199→        variant: "destructive",
   200→      });
   201→    } finally {
   202→      setSubmitting(false);
   203→    }
   204→  }
   205→
   206→  /* ---------- Styles par thème ---------- */
   207→  const inputBase = isLight
   208→    ? "min-h-[48px] w-full rounded-lg border border-[#e4e4e4] bg-white px-4 py-3 text-[15px] text-[#1c1c1c] placeholder:text-[#525252]/60 transition-colors focus:border-[#1c1c1c] focus:outline-none"
   209→    : isContact
   210→      ? "min-h-[50px] w-full rounded-md bg-white px-4 py-3 text-[15px] text-[#1c1c1c] placeholder:text-[#525252]/70 transition-shadow duration-200 focus:outline-none focus:shadow-[0_0_0_2px_rgba(228,228,228,0.55)]"
   211→      : "min-h-[48px] w-full rounded-lg border border-[#e4e4e4]/18 bg-[#1c1c1c] px-4 py-3 text-[15px] text-white placeholder:text-[#c8c8c8]/50 transition-colors focus:border-[#e4e4e4] focus:outline-none";
   212→  const labelBase = isLight
   213→    ? "mb-1.5 block text-sm font-medium text-[#1c1c1c]"
   214→    : "mb-1.5 block text-sm font-medium text-white";
   215→  const errorBase = "mt-1.5 text-xs text-[#ea384c]";
   216→
   217→  /* ---------- Confirmation (devis / audit) ---------- */
   218→  if (done) {
   219→    return (
   220→      <div
   221→        id={`${kind}-confirmation`}
   222→        className={
   223→          isLight
   224→            ? "rounded-2xl border border-[#e4e4e4] bg-white p-8 text-center sm:p-10"
   225→            : "rounded-2xl border border-[#e4e4e4]/40 bg-[#232323] p-8 text-center sm:p-10"
   226→        }
   227→      >
   228→        <span
   229→          className={
   230→            isLight
   231→              ? "mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-[#1c1c1c] text-white"
   232→              : "mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-[#e4e4e4] text-[#1c1c1c]"
   233→          }
   234→        >
   235→          <CheckCircle2 className="h-7 w-7" aria-hidden="true" />
   236→        </span>
   237→        <h3
   238→          className={`mt-6 font-display text-2xl font-semibold ${isLight ? "text-[#1c1c1c]" : "text-white"}`}
   239→        >
   240→          Votre demande a bien été enregistrée.
   241→        </h3>
   242→        <p className={`mx-auto mt-4 max-w-md text-[15px] leading-relaxed ${isLight ? "text-[#525252]" : "text-[#c8c8c8]"}`}>
   243→          {isQuote
   244→            ? "Merci pour votre confiance. Je reviens vers vous très rapidement pour préparer votre devis personnalisé."
   245→            : "Merci pour votre confiance. Je reviens vers vous très rapidement avec les premières observations sur votre site."}
   246→        </p>
   247→        <p className={`mt-4 text-sm ${isLight ? "text-[#525252]" : "text-[#c8c8c8]"}`}>
   248→          Un accusé de réception vous est adressé par email.
   249→        </p>
   250→        <div className="mt-8 flex flex-col items-center justify-center gap-3 sm:flex-row">
   251→          <button
   252→            type="button"
   253→            onClick={() => {
   254→              setDone(false);
   255→              mountedAt.current = Date.now();
   256→            }}
   257→            className={
   258→              isLight
   259→                ? "btn-fill min-h-[44px] rounded-full border border-[#1c1c1c]/25 bg-transparent px-6 py-3 text-sm font-medium text-[#1c1c1c] [&::before]:bg-[#e4e4e4]"
   260→                : "btn-fill min-h-[44px] rounded-full border border-[#e4e4e4]/30 px-6 py-3 text-sm font-medium text-white [&::before]:bg-[#3c3c3c]"
   261→            }
   262→          >
   263→            <span className="inline-flex items-center">Envoyer Une Autre Demande</span>
   264→          </button>
   265→        </div>
   266→        <p className={`mt-6 text-xs ${isLight ? "text-[#525252]/80" : "text-[#c8c8c8]/70"}`}>
   267→          Vous préférez avancer plus vite&nbsp;?{" "}
   268→          <a
   269→            href={SITE.calendly}
   270→            target="_blank"
   271→            rel="noopener noreferrer"
   272→            className={`font-medium underline underline-offset-4 ${isLight ? "text-[#1c1c1c]" : "text-[#e4e4e4]"}`}
   273→          >
   274→            Réservez un appel de 30 minutes
   275→          </a>{" "}
   276→          — {SITE.calendlyMention.toLowerCase()}
   277→        </p>
   278→      </div>
   279→    );
   280→  }
   281→
   282→  /* ---------- Formulaire ---------- */
   283→  return (
   284→    <form
   285→      ref={formRef}
   286→      onSubmit={handleSubmit}
   287→      noValidate
   288→      aria-label={
   289→        isContact
   290→          ? "Formulaire de contact — laissez un message"
   291→          : isQuote
   292→            ? "Formulaire de demande de devis personnalisé"
   293→            : "Formulaire de demande d'audit gratuit"
   294→      }
   295→      className={
   296→        isQuote
   297→          ? // Page claire (#/demander-un-devis) : champs posés directement
   298→            // sur le fond blanc, façon sheragency.
   299→            "relative"
   300→          : isContact
   301→            ? // Page sombre (#/contact) : champs posés directement sur le
   302→              // fond sombre, entrées blanches, façon sheragency.
   303→              "relative"
   304→            : "rounded-2xl border border-[#e4e4e4]/12 bg-[#232323] p-6 sm:p-8 lg:p-9"
   305→      }
   306→    >
   307→      {/* Piège anti-robots (invisible) */}
   308→      <input
   309→        type="text"
   310→        name="societe_verification"
   311→        value={form.honeypot}
   312→        onChange={set("honeypot")}
   313→        tabIndex={-1}
   314→        autoComplete="off"
   315→        aria-hidden="true"
   316→        className="pointer-events-none absolute left-[-9999px] h-0 w-0 opacity-0"
   317→      />
   318→
   319→      {/* ------- Variante « contact » : champs empilés (sheragency) ------- */}
   320→      {isContact ? (
   321→        <div className="flex flex-col gap-5">
   322→          <div>
   323→            <label htmlFor={`${kind}-field-name`} className={labelBase}>
   324→              Nom complet <span className={isLight ? "text-[#1c1c1c]" : "text-[#e4e4e4]"}>*</span>
   325→            </label>
   326→            <input
   327→              id={`${kind}-field-name`}
   328→              name="name"
   329→              autoComplete="name"
   330→              required
   331→              value={form.name}
   332→              onChange={set("name")}
   333→              aria-invalid={Boolean(errors.name)}
   334→              aria-describedby={errors.name ? `${kind}-error-name` : undefined}
   335→              className={inputBase}
   336→              placeholder="Votre nom et prénom"
   337→            />
   338→            {errors.name && (
   339→              <p id={`${kind}-error-name`} className={errorBase}>
   340→                {errors.name}
   341→              </p>
   342→            )}
   343→          </div>
   344→
   345→          <div>
   346→            <label htmlFor={`${kind}-field-phone`} className={labelBase}>
   347→              Téléphone <span className={isLight ? "text-[#1c1c1c]" : "text-[#e4e4e4]"}>*</span>
   348→            </label>
   349→            <input
   350→              id={`${kind}-field-phone`}
   351→              name="phone"
   352→              type="tel"
   353→              inputMode="tel"
   354→              autoComplete="tel"
   355→              required
   356→              value={form.phone}
   357→              onChange={set("phone")}
   358→              aria-invalid={Boolean(errors.phone)}
   359→              aria-describedby={errors.phone ? `${kind}-error-phone` : undefined}
   360→              className={inputBase}
   361→              placeholder="+229 …"
   362→            />
   363→            {errors.phone && (
   364→              <p id={`${kind}-error-phone`} className={errorBase}>
   365→                {errors.phone}
   366→              </p>
   367→            )}
   368→          </div>
   369→
   370→          <div>
   371→            <label htmlFor={`${kind}-field-email`} className={labelBase}>
   372→              Email <span className={isLight ? "text-[#1c1c1c]" : "text-[#e4e4e4]"}>*</span>
   373→            </label>
   374→            <input
   375→              id={`${kind}-field-email`}
   376→              name="email"
   377→              type="email"
   378→              inputMode="email"
   379→              autoComplete="email"
   380→              required
   381→              value={form.email}
   382→              onChange={set("email")}
   383→              aria-invalid={Boolean(errors.email)}
   384→              aria-describedby={errors.email ? `${kind}-error-email` : undefined}
   385→              className={inputBase}
   386→              placeholder="vous@exemple.com"
   387→            />
   388→            {errors.email && (
   389→              <p id={`${kind}-error-email`} className={errorBase}>
   390→                {errors.email}
   391→              </p>
   392→            )}
   393→          </div>
   394→
   395→          <div>
   396→            <label htmlFor={`${kind}-field-company`} className={labelBase}>
   397→              Entreprise <span className={isLight ? "text-[#1c1c1c]" : "text-[#e4e4e4]"}>*</span>
   398→            </label>
   399→            <input
   400→              id={`${kind}-field-company`}
   401→              name="company"
   402→              autoComplete="organization"
   403→              required
   404→              value={form.company}
   405→              onChange={set("company")}
   406→              aria-invalid={Boolean(errors.company)}
   407→              aria-describedby={errors.company ? `${kind}-error-company` : undefined}
   408→              className={inputBase}
   409→              placeholder="Nom de votre entreprise ou activité"
   410→            />
   411→            {errors.company && (
   412→              <p id={`${kind}-error-company`} className={errorBase}>
   413→                {errors.company}
   414→              </p>
   415→            )}
   416→          </div>
   417→
   418→          <div>
   419→            <label htmlFor={`${kind}-field-discovery`} className={labelBase}>
   420→              Comment m'avez-vous découvert&nbsp;?{" "}
   421→              <span className={isLight ? "text-[#1c1c1c]" : "text-[#e4e4e4]"}>*</span>
   422→            </label>
   423→            <select
   424→              id={`${kind}-field-discovery`}
   425→              name="discovery"
   426→              required
   427→              value={form.discovery}
   428→              onChange={set("discovery")}
   429→              aria-invalid={Boolean(errors.discovery)}
   430→              className={`${inputBase} appearance-none`}
   431→            >
   432→              <option value="">— Choisir —</option>
   433→              {DISCOVERY_SOURCES.map((s) => (
   434→                <option key={s} value={s}>
   435→                  {s}
   436→                </option>
   437→              ))}
   438→            </select>
   439→            {errors.discovery && (
   440→              <p id={`${kind}-error-discovery`} className={errorBase}>
   441→                {errors.discovery}
   442→              </p>
   443→            )}
   444→          </div>
   445→
   446→          <div>
   447→            <label htmlFor={`${kind}-field-message`} className={labelBase}>
   448→              Comment puis-je vous aider&nbsp;?{" "}
   449→              <span className={isLight ? "text-[#1c1c1c]" : "text-[#e4e4e4]"}>*</span>
   450→            </label>
   451→            <textarea
   452→              id={`${kind}-field-message`}
   453→              name="message"
   454→              required
   455→              rows={5}
   456→              value={form.message}
   457→              onChange={set("message")}
   458→              aria-invalid={Boolean(errors.message)}
   459→              aria-describedby={errors.message ? `${kind}-error-message` : undefined}
   460→              className={`${inputBase} min-h-[140px] resize-y py-3.5`}
   461→              placeholder="Décrivez votre demande…"
   462→            />
   463→            {errors.message && (
   464→              <p id={`${kind}-error-message`} className={errorBase}>
   465→                {errors.message}
   466→              </p>
   467→            )}
   468→          </div>
   469→        </div>
   470→      ) : (
   471→        /* ------- Variante « devis » / « audit » : grille deux colonnes ------- */
   472→        <div className="grid gap-5 sm:grid-cols-2">
   473→          <div>
   474→            <label htmlFor={`${kind}-field-name`} className={labelBase}>
   475→              Nom complet{" "}
   476→              <span className={isLight ? "text-[#1c1c1c]" : "text-[#e4e4e4]"}>*</span>
   477→            </label>
   478→            <input
   479→              id={`${kind}-field-name`}
   480→              name="name"
   481→              autoComplete="name"
   482→              required
   483→              value={form.name}
   484→              onChange={set("name")}
   485→              aria-invalid={Boolean(errors.name)}
   486→              aria-describedby={errors.name ? `${kind}-error-name` : undefined}
   487→              className={inputBase}
   488→              placeholder="Votre nom et prénom"
   489→            />
   490→            {errors.name && (
   491→              <p id={`${kind}-error-name`} className={errorBase}>
   492→                {errors.name}
   493→              </p>
   494→            )}
   495→          </div>
   496→
   497→          <div>
   498→            <label htmlFor={`${kind}-field-email`} className={labelBase}>
   499→              Email <span className={isLight ? "text-[#1c1c1c]" : "text-[#e4e4e4]"}>*</span>
   500→            </label>
   501→            <input

   514→            />

   523→            <label htmlFor={`${kind}-field-phone`} className={labelBase}>
   524→              Téléphone <span className={isLight ? "text-[#525252]" : "text-[#c8c8c8]/60"}>(optionnel)</span>
